export function buildCustomFieldsSchema(defs: FieldDef[]): z.ZodType {
  const shape: Record<string, z.ZodType> = {};
  for (const def of defs) {

    shape[def.key] = def.required === true ? base : base.optional();
  }
  return z.object(shape).strict();
}
